cmake_minimum_required(VERSION 3.24)
project(discord_voice_proxy_doh LANGUAGES CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This installer targets Windows x64.")
endif()
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
find_package(CURL CONFIG REQUIRED)
find_path(DETOURS_INCLUDE_DIR detours/detours.h REQUIRED)
find_library(DETOURS_RELEASE detours PATHS "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(DETOURS_DEBUG detours PATHS "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/debug/lib" NO_DEFAULT_PATH REQUIRED)
add_library(detours INTERFACE)
target_link_libraries(detours INTERFACE optimized "${DETOURS_RELEASE}" debug "${DETOURS_DEBUG}")
set(FORCE_PROXY_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/force-proxy/src" CACHE PATH "force-proxy source directory")
add_library(force-proxy SHARED
  "${FORCE_PROXY_SOURCE}/dllmain.cpp" "${FORCE_PROXY_SOURCE}/hooks.cpp"
  "${FORCE_PROXY_SOURCE}/proxy.cpp" "${FORCE_PROXY_SOURCE}/dns_hooks.cpp"
  "${FORCE_PROXY_SOURCE}/dns_wire.cpp")
target_include_directories(force-proxy PRIVATE "${DETOURS_INCLUDE_DIR}")
target_link_libraries(force-proxy PRIVATE CURL::libcurl detours ws2_32 normaliz)
add_library(DWrite SHARED proxy-dll/dllmain.cpp proxy-dll/proxy.asm proxy-dll/browser_proxy.cpp)
target_include_directories(DWrite PRIVATE "${DETOURS_INCLUDE_DIR}")
target_link_libraries(DWrite PRIVATE detours)
set_target_properties(force-proxy DWrite PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/payload")
enable_testing()
add_executable(dns_wire_test "${FORCE_PROXY_SOURCE}/../tests/dns_wire_test.cpp" "${FORCE_PROXY_SOURCE}/dns_wire.cpp")
target_include_directories(dns_wire_test PRIVATE "${FORCE_PROXY_SOURCE}")
add_test(NAME dns_wire COMMAND dns_wire_test)
add_executable(proxy_dns_probe "${FORCE_PROXY_SOURCE}/../tests/proxy_dns_probe.cpp")
target_link_libraries(proxy_dns_probe PRIVATE ws2_32)
set_target_properties(proxy_dns_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/payload")
add_executable(dns_perf_test "${FORCE_PROXY_SOURCE}/../tests/dns_perf_test.cpp")
target_link_libraries(dns_perf_test PRIVATE ws2_32 psapi)
set_target_properties(dns_perf_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/payload")
